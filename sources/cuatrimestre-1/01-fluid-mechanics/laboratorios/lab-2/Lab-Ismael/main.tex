\begin{figure}[H]
\centering
\includegraphics[width=0.9\linewidth]{fig_neumann.pdf}
\caption{$f(x,t)$ with Neumann boundary conditions ($\partial_x f=0$ at $x=\pm10$) at $t=1$, $5$ and $15$. Parameters: $U=1$, $\alpha=0.1$ ($k=0.5$). The initial pulse starts at $x=-2$, and the flow carries it to the right at speed $U$: its peak is at $x\simeq-1$ for $t=1$ and at $x\simeq3$ for $t=5$. Its height drops a little ($\max f=0.965$ and $0.86$) because diffusion spreads it out. At $t=15$ the pulse has reached the right end, where the zero-slope condition makes the curve flat ($f\simeq0.27$ at $x=10$) and the rest of the pulse leaves the domain.}
\label{fig:neumann}
\end{figure}

\begin{figure}[H]
\centering
\includegraphics[width=0.9\linewidth]{fig_periodic.pdf}
\caption{$f(x,t)$ with periodic boundary conditions ($x=-10$ and $x=10$ are the same point) at $t=1$, $5$ and $15$. Parameters: $U=1$, $\alpha=0.1$ ($k=0.5$). The curves at $t=1$ and $t=5$ are the same as in Fig.~\ref{fig:neumann}, since the pulse has not reached the boundaries yet. At $t=15$ the pulse would be at $x=13$, outside the domain, so it comes back in from the left and its peak appears at $x\simeq-7$ ($13-20$), with $\max f=0.71$. Nothing is lost at the boundaries, so the pulse only gets lower and wider because of diffusion.}
\label{fig:periodic}
\end{figure}
