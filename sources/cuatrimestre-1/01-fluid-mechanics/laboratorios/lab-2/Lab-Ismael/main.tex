\begin{figure}[H]
\centering
\includegraphics[width=0.9\linewidth]{fig_neumann.pdf}
\caption{$f(x,t)$ with Neumann boundary conditions ($\partial_x f=0$ at $x=\pm10$) at $t=1$, $5$ and $15$. Parameters: $U=1$, $\alpha=0.1$ ($k=0.5$). The pulse starts centred at $x=-2$ and the flow pushes it to the right at speed $U$, so its peak is at $x\simeq-1$ for $t=1$ and at $x\simeq3$ for $t=5$. On the way it gets a bit lower ($\max f=0.965$ and $0.86$), since diffusion keeps spreading it out. By $t=15$ it has reached the right end. The highest value is now right at the wall ($f\simeq0.27$ at $x=10$) and most of the pulse is already gone: setting the slope to zero at the wall does not stop the flow from carrying $f$ out of the domain.}
\label{fig:neumann}
\end{figure}

\begin{figure}[H]
\centering
\includegraphics[width=0.9\linewidth]{fig_periodic.pdf}
\caption{$f(x,t)$ with periodic boundary conditions ($x=-10$ and $x=10$ are the same point) at $t=1$, $5$ and $15$. Parameters: $U=1$, $\alpha=0.1$ ($k=0.5$). At $t=1$ and $t=5$ the curves look almost the same as in Fig.~\ref{fig:neumann}, because the pulse has not hit either boundary yet. The difference shows up at $t=15$. By then the pulse should be at $x=13$, which is outside the domain, so it re-enters from the left and its peak shows up at $x\simeq-7$ ($13-20$), with $\max f=0.71$. Here nothing leaves through the boundaries, so the only thing diffusion does is make the pulse lower and wider.}
\label{fig:periodic}
\end{figure}
